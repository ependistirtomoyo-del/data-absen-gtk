Petunjuk menghubungkan fitur SIMPAN & EDIT langsung ke Spreadsheet

Informasi Penyimpanan Direct Write:

Spreadsheet TSV Google Sheet yang dipublikasikan secara default bersifat Read-Only. Agar fitur **Tambah Data** dan **Edit Data** dari web ini tersimpan langsung secara otomatis di Google Sheet Anda, Anda dapat menempelkan Google Apps Script Web App di bawah ini ke Spreadsheet Anda.

Langkah Mudah (Hanya 1 Menit):

  1. Buka file Google Sheet Anda di browser.
  2. Klik menu Extensions (Ekstensi) > Apps Script.
  3. Hapus isi kode yang ada, lalu salin dan tempel (paste) kode di bawah ini:
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    if (data.action === 'add') {
      sheet.appendRow(data.row);
      return ContentService.createTextOutput(JSON.stringify({result: 'success', message: 'Data ditambahkan'}))
        .setMimeType(ContentService.MimeType.JSON);
    } 
    else if (data.action === 'edit') {
      var rowIndex = parseInt(data.rowIndex) + 1; // 1-based index (header = row 1)
      var rowData = data.row;
      for (var col = 0; col < rowData.length; col++) {
        sheet.getRange(rowIndex, col + 1).setValue(rowData[col]);
      }
      return ContentService.createTextOutput(JSON.stringify({result: 'success', message: 'Data diperbarui'}))
        .setMimeType(ContentService.MimeType.JSON);
    }
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({result: 'error', error: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
